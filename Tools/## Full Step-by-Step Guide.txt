## What the App Does

```
User types topic → clicks Generate
         ↓
Slide content created automatically
         ↓
Audio MP3s generated for each slide
         ↓
PowerPoint file created with notes
         ↓
Download PPT + Audio ZIP from browser
```

```
my_ppt_app/
├── app.py          ← the web app

```

# ============================================
# GENERATE AUDIO FILES
# ============================================

async def _generate_audio_async(slides: list, audio_dir: str):
    """Async function to generate all audio files."""
    for i, slide in enumerate(slides):
        filename = os.path.join(audio_dir, f"slide_{i+1}.mp3")
        communicate = edge_tts.Communicate(
            slide["text"],
            voice="en-US-AriaNeural"
        )
        await communicate.save(filename)

def generate_audio(slides: list, audio_dir: str = "audio"):
    """Generate MP3 audio for each slide."""
    os.makedirs(audio_dir, exist_ok=True)
    asyncio.run(_generate_audio_async(slides, audio_dir))


# ============================================
# CREATE POWERPOINT
# ============================================

def create_ppt(slides: list, audio_dir: str, output_path: str):
    """Create a PowerPoint presentation from slide data."""

    prs = Presentation()
    total_duration = 0

    for i, slide_data in enumerate(slides):

        slide = prs.slides.add_slide(prs.slide_layouts[1])

        # Title
        title_shape = slide.shapes.title
        title_shape.text = slide_data["title"]
        title_shape.text_frame.paragraphs[0].font.size = Pt(28)
        title_shape.text_frame.paragraphs[0].font.bold = True
        title_shape.text_frame.paragraphs[0].font.color.rgb = RGBColor(0x2C, 0x3E, 0x50)

        # Content body
        body = slide.placeholders[1]
        tf = body.text_frame
        tf.clear()
        tf.word_wrap = True

        lines = [l.strip() for l in slide_data["text"].strip().split("\n") if l.strip()]

        for idx, line in enumerate(lines):
            p = tf.paragraphs[0] if idx == 0 else tf.add_paragraph()
            p.text = line
            p.font.size = Pt(17)
            p.font.color.rgb = RGBColor(0x2C, 0x3E, 0x50)
            p.space_after = Pt(6)

        # Add duration to notes
        audio_file = os.path.join(audio_dir, f"slide_{i+1}.mp3")
        if os.path.exists(audio_file):
            audio = MP3(audio_file)
            duration = round(audio.info.length, 2)
            total_duration += duration
            notes = slide.notes_slide.notes_text_frame
            notes.text = f"Audio: slide_{i+1}.mp3 | Duration: {duration} seconds"

    prs.save(output_path)
    return total_duration


# ============================================
# FULL PIPELINE
# ============================================

def run_pipeline(topic: str, num_slides: int, audio_dir: str, output_ppt: str):
    """
    Full pipeline:
    1. Generate slide content
    2. Generate audio
    3. Create PPT
    Returns total duration in seconds.
    """
    slides = generate_slides(topic, num_slides)
    generate_audio(slides, audio_dir)
    total_duration = create_ppt(slides, audio_dir, output_ppt)
    return total_duration, slides
